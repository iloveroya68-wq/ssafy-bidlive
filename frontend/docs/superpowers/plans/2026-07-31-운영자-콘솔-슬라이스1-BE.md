# 운영자 콘솔 슬라이스 1 (BE) Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 운영자가 신고 접수된 경매를 경매별로 조회할 수 있는 `GET /api/v1/admin/reports`와, 가입한 회원을 `ADMIN`으로 승격하는 환경변수 기반 초기화를 붙인다.

```java
package com.ssafy.bidlive.common.config;

import java.util.Arrays;
import java.util.List;

import jakarta.persistence.EntityManager;
import org.springframework.beans.factory.annotation.Value;
import org.springframework.boot.ApplicationArguments;
import org.springframework.boot.ApplicationRunner;
import org.springframework.stereotype.Component;
import org.springframework.transaction.annotation.Transactional;

/**
 * `ADMIN_MEMBER_IDS`에 적힌 회원을 부팅 시 `ADMIN`으로 올린다.
 *
 * <p><b>시드로 관리자 계정을 만들 수는 없다.</b> 로그인 경로가 카카오 OAuth뿐이고
 * `social_account`에 실제 `provider_user_id`가 붙어야 들어올 수 있다 — 비밀번호 개념이 없다.
 * 그래서 「계정 생성」이 아니라 <b>「이미 가입한 회원의 승격」</b>만 가능하다.
 *
 * <p>Flyway 마이그레이션에 `UPDATE ... WHERE member_id = N`을 넣지 않는 이유: 특정 개인의 id가
 * git 히스토리에 영구히 남고, <b>로컬 DB의 `member_id`는 사람마다 달라서 다른 사람 환경에서
 * 엉뚱한 회원이 ADMIN이 된다.</b>
 *
 * <p>⚠️ 이 방식은 env 파일이 곧 권한이다. 지금 규모에서는 받아들이고, 회원 관리 화면이 생기면
 * 그쪽으로 옮긴다.
 */
@Component
public class AdminMemberRoleInitializer implements ApplicationRunner {

	private static final String PROMOTE = """
		UPDATE member
		SET role = 'ADMIN'
		WHERE member_id IN (:memberIds)
			AND role <> 'ADMIN'
		""";

	private final EntityManager entityManager;
	private final String configuredIds;

	public AdminMemberRoleInitializer(
		EntityManager entityManager,
		@Value("${ADMIN_MEMBER_IDS:}") String configuredIds
	) {
		this.entityManager = entityManager;
		this.configuredIds = configuredIds;
	}

	@Override
	public void run(ApplicationArguments args) {
		promote();
	}

	/** 값이 비었거나 쓸 수 있는 id가 없으면 아무것도 하지 않는다. 재실행에 안전하다. */
	@Transactional
	public void promote() {
		List<Long> memberIds = parse(configuredIds);
		if (memberIds.isEmpty()) {
			return;
		}
		entityManager.createNativeQuery(PROMOTE)
			.setParameter("memberIds", memberIds)
			.executeUpdate();
	}

	/** 숫자가 아니거나 0 이하인 항목은 조용히 버린다 — 오타 하나로 부팅을 막지 않는다. */
	private static List<Long> parse(String raw) {
		if (raw == null || raw.isBlank()) {
			return List.of();
		}
		return Arrays.stream(raw.split(","))
			.map(String::trim)
			.filter(value -> !value.isEmpty())
			.map(AdminMemberRoleInitializer::toPositiveLong)
			.filter(value -> value != null)
			.toList();
	}

	private static Long toPositiveLong(String value) {
		try {
			long parsed = Long.parseLong(value);
			return parsed > 0 ? parsed : null;
		} catch (NumberFormatException exception) {
			return null;
		}
	}
}
```

```java
package com.ssafy.bidlive.report.domain;

import java.nio.charset.StandardCharsets;
import java.time.Instant;
import java.time.format.DateTimeParseException;
import java.util.Base64;

/**
 * 신고 접수 경매 목록의 최신순 키셋 페이징 커서.
 *
 * <p>정렬 키가 시각 하나면 같은 시각의 행에서 페이지가 겹치거나 빠진다 —
 * {@code auctionId}를 타이브레이커로 함께 싣는다(`DealListCursor`와 같은 규약).
 */
public record ReportedAuctionCursor(Instant latestReportedAt, long auctionId) {

	private static final char DELIMITER = '|';

	public ReportedAuctionCursor {
		if (latestReportedAt == null || auctionId <= 0) {
			throw new IllegalArgumentException("reported auction cursor");
		}
	}

	public String encode() {
		String payload = latestReportedAt + String.valueOf(DELIMITER) + auctionId;
		return Base64.getUrlEncoder()
			.withoutPadding()
			.encodeToString(payload.getBytes(StandardCharsets.UTF_8));
	}

	/** @throws IllegalArgumentException 커서 형식이나 값이 유효하지 않을 때 */
	public static ReportedAuctionCursor decode(String encoded) {
		try {
			String payload = new String(
				Base64.getUrlDecoder().decode(encoded),
				StandardCharsets.UTF_8
			);
			int delimiter = payload.lastIndexOf(DELIMITER);
			if (delimiter <= 0 || delimiter == payload.length() - 1) {
				throw new IllegalArgumentException("reported auction cursor");
			}
			return new ReportedAuctionCursor(
				Instant.parse(payload.substring(0, delimiter)),
				Long.parseLong(payload.substring(delimiter + 1))
			);
		} catch (DateTimeParseException | NumberFormatException exception) {
			throw new IllegalArgumentException("reported auction cursor", exception);
		}
	}
}
```

```java
package com.ssafy.bidlive.report.infrastructure.persistence;

import java.sql.Timestamp;
import java.time.Instant;
import java.util.LinkedHashMap;
import java.util.List;
import java.util.Map;

import com.ssafy.bidlive.report.application.AdminReportQueryPersistence;
import com.ssafy.bidlive.report.application.ReportedAuction;
import com.ssafy.bidlive.report.domain.ReportReasonCount;
import com.ssafy.bidlive.report.domain.ReportedAuctionCursor;
import jakarta.persistence.EntityManager;
import jakarta.persistence.Query;
import org.springframework.stereotype.Repository;

/**
 * 신고를 경매별로 집계해 읽는다.
 *
 * <p><b>두 쿼리로 나눈다.</b> 사유별 카운트를 같은 쿼리에 조인하면 경매 행이 사유 수만큼 곱해져
 * {@code reportCount}와 페이지 크기가 함께 틀어진다.
 *
 * <p>커서 조건이 {@code WHERE}가 아니라 {@code HAVING}인 이유: 정렬 키가 집계값
 * {@code MAX(r.created_at)}이라 그룹화 전에는 값이 없다.
 */
@Repository
public class JpaAdminReportQueryAdapter implements AdminReportQueryPersistence {

	private static final String FIND_REPORTED_AUCTIONS = """
		SELECT
			ar.auction_room_id,
			p.title,
			(
				SELECT thumbnail.image_url
				FROM product_image thumbnail
				WHERE thumbnail.product_id = p.product_id
					AND thumbnail.moderation_status <> 'BLOCKED'
				ORDER BY thumbnail.is_thumbnail DESC, thumbnail.sort_order ASC
				LIMIT 1
			),
			m.nickname,
			ar.status,
			COUNT(*),
			MAX(r.created_at)
		FROM report r
		JOIN auction_room ar ON ar.auction_room_id = r.auction_room_id
		JOIN product p ON p.product_id = ar.product_id
		JOIN member m ON m.member_id = p.seller_member_id
		WHERE r.auction_room_id IS NOT NULL
			AND ar.deleted_at IS NULL
			AND (:includeEnded = TRUE OR ar.status <> 'ENDED')
		GROUP BY ar.auction_room_id, p.product_id, p.title, m.nickname, ar.status
		HAVING (
			CAST(:cursorAt AS TIMESTAMPTZ) IS NULL
			OR MAX(r.created_at) < CAST(:cursorAt AS TIMESTAMPTZ)
			OR (
				MAX(r.created_at) = CAST(:cursorAt AS TIMESTAMPTZ)
				AND ar.auction_room_id < :cursorAuctionId
			)
		)
		ORDER BY MAX(r.created_at) DESC, ar.auction_room_id DESC
		LIMIT :limit
		""";

	private static final String FIND_REASON_COUNTS = """
		SELECT
			r.auction_room_id,
			r.reason_code,
			COUNT(*)
		FROM report r
		WHERE r.auction_room_id IN (:auctionIds)
		GROUP BY r.auction_room_id, r.reason_code
		ORDER BY r.auction_room_id, COUNT(*) DESC, r.reason_code ASC
		""";

	private final EntityManager entityManager;

	public JpaAdminReportQueryAdapter(EntityManager entityManager) {
		this.entityManager = entityManager;
	}

	@Override
	public List<ReportedAuction> findReportedAuctions(
		boolean includeEnded,
		ReportedAuctionCursor cursor,
		int limit
	) {
		Query query = entityManager.createNativeQuery(FIND_REPORTED_AUCTIONS)
			.setParameter("includeEnded", includeEnded)
			.setParameter("cursorAt", cursor == null ? null : cursor.latestReportedAt().toString())
			.setParameter("cursorAuctionId", cursor == null ? 0L : cursor.auctionId())
			.setParameter("limit", limit);
		return query.getResultList().stream()
			.map(Object[].class::cast)
			.map(row -> new ReportedAuction(
				((Number) row[0]).longValue(),
				(String) row[1],
				(String) row[2],
				(String) row[3],
				(String) row[4],
				((Number) row[5]).intValue(),
				toInstant(row[6]),
				List.of()
			))
			.toList();
	}

	@Override
	public Map<Long, List<ReportReasonCount>> findReasonCounts(List<Long> auctionIds) {
		if (auctionIds.isEmpty()) {
			return Map.of();
		}
		Map<Long, List<ReportReasonCount>> grouped = new LinkedHashMap<>();
		entityManager.createNativeQuery(FIND_REASON_COUNTS)
			.setParameter("auctionIds", auctionIds)
			.getResultList().stream()
			.map(Object[].class::cast)
			.forEach(row -> grouped
				.computeIfAbsent(((Number) row[0]).longValue(), key -> new java.util.ArrayList<>())
				.add(new ReportReasonCount(
					(String) row[1],
					((Number) row[2]).intValue()
				)));
		return grouped;
	}

	/** 드라이버가 `Timestamp`를 주기도 하고 `Instant`를 주기도 한다 — 둘 다 받는다. */
	private static Instant toInstant(Object value) {
		if (value instanceof Timestamp timestamp) {
			return timestamp.toInstant();
		}
		return (Instant) value;
	}
}
```

```java
package com.ssafy.bidlive.report.application;

import java.util.List;
import java.util.Map;

import com.ssafy.bidlive.report.domain.ReportReasonCount;
import com.ssafy.bidlive.report.domain.ReportedAuctionCursor;
import org.springframework.stereotype.Service;
import org.springframework.transaction.annotation.Transactional;

/**
 * ADMIN-01 운영자 신고 목록 조회.
 *
 * <p>다음 장 존재는 <b>size + 1건을 읽어</b> 판정한다 — COUNT를 따로 치면 두 쿼리가 서로 다른
 * 스냅샷을 보게 되어 「다음 장이 있다는데 비어 있는」 경계가 생긴다.
 */
@Service
public class AdminReportQueryService {

	private static final int DEFAULT_SIZE = 20;
	private static final int MAX_SIZE = 100;

	private final AdminReportQueryPersistence persistence;

	public AdminReportQueryService(AdminReportQueryPersistence persistence) {
		this.persistence = persistence;
	}

	@Transactional(readOnly = true)
	public ReportedAuctionPage getReportedAuctions(
		boolean includeEnded,
		ReportedAuctionCursor cursor,
		int size
	) {
		int normalized = normalizeSize(size);
		List<ReportedAuction> fetched =
			persistence.findReportedAuctions(includeEnded, cursor, normalized + 1);

		boolean hasNext = fetched.size() > normalized;
		List<ReportedAuction> page = hasNext ? fetched.subList(0, normalized) : fetched;

		Map<Long, List<ReportReasonCount>> reasons = persistence.findReasonCounts(
			page.stream().map(ReportedAuction::auctionId).toList()
		);
		List<ReportedAuction> items = page.stream()
			.map(auction -> auction
				.withReasonCounts(reasons.getOrDefault(auction.auctionId(), List.of()))
				.withMaskedSeller())
			.toList();

		return new ReportedAuctionPage(
			items,
			hasNext ? nextCursorOf(items) : null,
			hasNext
		);
	}

	/** 명세 「기본 20, 최대 100」. 0 이하는 기본값으로 되돌린다 */
	private static int normalizeSize(int size) {
		if (size <= 0) {
			return DEFAULT_SIZE;
		}
		return Math.min(size, MAX_SIZE);
	}

	/** 커서는 **잘라낸 뒤 마지막 행** 기준이다. 여분 행으로 만들면 한 건이 건너뛰어진다. */
	private static String nextCursorOf(List<ReportedAuction> items) {
		ReportedAuction last = items.get(items.size() - 1);
		return new ReportedAuctionCursor(last.latestReportedAt(), last.auctionId()).encode();
	}
}
```

```java
package com.ssafy.bidlive.report.api;

import com.ssafy.bidlive.report.api.dto.AdminReportListResponse;
import com.ssafy.bidlive.report.application.AdminReportQueryService;
import com.ssafy.bidlive.report.domain.ReportedAuctionCursor;
import org.springframework.http.HttpStatus;
import org.springframework.web.bind.annotation.GetMapping;
import org.springframework.web.bind.annotation.RequestMapping;
import org.springframework.web.bind.annotation.RequestParam;
import org.springframework.web.bind.annotation.RestController;
import org.springframework.web.server.ResponseStatusException;

/**
 * ADMIN-01 운영자 신고 목록.
 *
 * <p>권한은 `SecurityConfig`가 경로 단위로 건다 — 이 레포는 메서드 시큐리티를 켜지 않는다.
 */
@RestController
@RequestMapping("/api/v1/admin/reports")
public class AdminReportController {

	private final AdminReportQueryService queryService;

	public AdminReportController(AdminReportQueryService queryService) {
		this.queryService = queryService;
	}

	@GetMapping
	public AdminReportListResponse getReports(
		@RequestParam(required = false, defaultValue = "false") boolean includeEnded,
		@RequestParam(required = false) String cursor,
		// int로 받으면 `size=abc`가 Spring 타입 변환에서 터져 이 봉투를 벗어난다.
		// 문자열로 받아 다른 파라미터와 같은 경로로 400을 낸다.
		@RequestParam(required = false) String size
	) {
		return AdminReportListResponse.from(queryService.getReportedAuctions(
			includeEnded,
			parseCursor(cursor),
			parseSize(size)
		));
	}

	private static ReportedAuctionCursor parseCursor(String cursor) {
		if (cursor == null || cursor.isBlank()) {
			return null;
		}
		try {
			return ReportedAuctionCursor.decode(cursor);
		} catch (IllegalArgumentException exception) {
			throw new ResponseStatusException(
				HttpStatus.BAD_REQUEST, "INVALID_CURSOR", exception
			);
		}
	}

	/** 없으면 0을 돌려 서비스의 기본값(20) 분기로 보낸다. */
	private static int parseSize(String size) {
		if (size == null || size.isBlank()) {
			return 0;
		}
		try {
			return Integer.parseInt(size);
		} catch (NumberFormatException exception) {
			throw new ResponseStatusException(
				HttpStatus.BAD_REQUEST, "INVALID_SIZE", exception
			);
		}
	}
}
```
